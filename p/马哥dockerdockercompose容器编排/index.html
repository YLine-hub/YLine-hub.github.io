vim docker-compose.yml
cat docker-compose.yml
services:
  service-nginx-web:
    image: centos7-nginx:1.6.1.v2
    container_name: nginx-web
    volumes:
      - /data/nginx:/apps/nginx/html # Specify a volume to mount the host directory /data/nginx to /apps/nginx/html in the container
    expose:
      - 80
      - 443
    ports:
      - "80:80"
      - "443:443"
    

  service-tomcat-app1:
    image: tomcat-web:app1
    container_name: tomcat-app1
    expose:
      - 8080
    ports:
      - "8081:8080"
    command: /start.sh

  service-tomcat-app2:
    image: tomcat-web:app1
    container_name: tomcat-app2
    expose:
      - 8080
    ports:
      - "8082:8080"
    command: /start.sh

    <role rolename="admin-gui"/>
    <role rolename="admin-script"/>

      用户定义：
      用户名：admin
      密码：admin123（请根据实际情况修改）
      角色：拥有所有管理权限

    <user username="admin" password="admin123"
          roles="manager-gui,manager-script,manager-jmx,manager-status,admin-gui,admin-script"/>
