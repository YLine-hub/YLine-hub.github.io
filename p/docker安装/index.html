# 新增docker.service文件
vim /usr/lib/systemd/system/docker.service
# 按下面的内容填写
----
[Unit]
Description=Docker Application Container Engine
Documentation=https://docs.docker.com
After=network-online.target firewalld.service
Wants=network-online.target
[Service]
Type=notify
ExecStart=/usr/bin/dockerd
ExecReload=/bin/kill -s HUP $MAINPID
LimitNOFILE=infinity
LimitNPROC=infinity
TimeoutStartSec=0
Delegate=yes
KillMode=process
Restart=on-failure
StartLimitBurst=3
StartLimitInterval=60s
[Install]
WantedBy=multi-user.target
----

问题

ubuntu 启动后，使用exit命令退出，再使用start启动，有时候无法启动，有时候能够正常启动

原因：创建ubuntu容器时，有无使用交互模式，如果使用了能启动，没有使用的话无法启动

  • 情况一：使用交互模式启动
1
2
3
4
5
6
7

[root@kylin-v11 ~]# docker run -it ubuntu bash
root@29d9e137b748:/# exit 
exit
[root@kylin-v11 ~]# docker start 29d9
29d9
[root@kylin-v11 ~]# docker ps 
CONTAINER ID   IMAGE     COMMAND       CREATED          STATUS         PORTS     NAMES
29d9e137b748   ubuntu    "bash"        21 seconds ago   Up 8 seconds             inspiring_jackson
  • 情况二：没有使用交互模式

Docker Registry

Pull the Docker Registry 

1

docker pull registry

Run the Docker Resgitry 

1

docker run -d -p 5000:5000 --restart always --name registry registry

Details of Docker Registry

List all the images

1

curl -XGET http://ip:5000/v2/_catalog

List the tags of the specified image

1

curl -XGET http://ip:5000/v2/<image_name>/tags/list

Native basic auth

Create a password file with one entry for the user testuser,with password testpassword;

1

docker run --entrypoint htpasswd httpd:2 -Bbn testuser testpassword > /yline/auth/htpasswd

Stop the registry

1

docker container stop registry

Start the registry with basic authentication

1
2
3
4
5
6
7
8
9

docker run -d \
    -p 5000:5000 \
    --name registry \
    -v /yline/myregistry:/tmp/registry \
    -v /yline/auth:/auth \
    -e "REGISTRY_AUTH=htpasswd" \
    -e REGISTRY_AUTH_HTPASSWD_PATH=/auth/htpasswd \
    -e REGISTRY_AUTH_HTPASSWD_REALM="Registry Realm" \
    registry

How do I see the command used to create a container in Docker

  • Using docker inspect
1

docker inspect [CONTAINER_ID]
Inspect Field Path docker run argument sample value
.Config.Image Image Name regitry:3
.Name –name /registry
.Config.Hostname –hostname 04e1904c7da5
.HostConfig.Binds -v/–volume null
.HostConfig.PortBindings -p/-publish map[5000/tcp:[{invalid IP 5000}]]
.HostConfig.RestartPolicy.Name –restart always

Combine these arguments in docker inspect

1

docker run --name=registry --hostname=04e1904c7da5 -p 5000:5000 --restart=always registry:3
