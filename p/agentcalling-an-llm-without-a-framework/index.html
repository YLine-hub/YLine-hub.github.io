import os

from dotenv import load_dotenv
from openai import OpenAI

# 1.Load environment variables
load_dotenv()

# 2.Create an LLM client object
client = OpenAI(
    api_key=os.getenv("BAILIAN_API_KEY"),
    base_url=os.getenv("BAILIAN_BASE_URL")
)

# 3. Call the LLM for text-to-text generation
resp = client.chat.completions.create(
    model="qwen3.8-max",    # Model name，use whichever is cheaper
    messages=[
        {'role':'user','content':"把'你好'翻译成意大利语"}    # User prompt, i.e., the user's question
    ]
)

# 4. The LLM returns a result, and then parse the result
print(resp)
print(resp.choices[0].message.content)

from dotenv import load_dotenv
from openai import OpenAI

# 1. Load environment variables
load_dotenv()

# 2. Create an LLM client object
client = OpenAI()   # Omit the default parameter

# 3. Call the LLM for text-to-text generation
resp = client.responses.create(
    model="qwen3.8-max",
    input="中国国内今天发生了哪些大事？",
    tools=[
        {"type":"web_search"}   # Enable web search
    ]
)

# 4. The LLM returns a result, and then parse the result
print(resp)
print(resp.output_text)
