Create a Tomcat web image using an Ubuntu 22 container in docker

Apache-tomcat-8.5.50.tar.gz download link

  • Apache-tomcat-8.5.50.tar.gz

Complete operation process

Start an Ubuntu 22.04 container 

1

docker run -it --name tomcat-builder ubuntu:22.04 /bin/bash

Update the system and install necessary tools

# Update the package list
apt update

# Install tools such as wget, curl, and vim
apt install -y wget curl vim openjdk-8-jdk net-tools

# Verify that JDK is installed
java -version

# Create the Tomcat installation directory
mkdir -p /apps/tomcat
cd /apps/tomcat

# Create the web direcotry
mkdir -p /data/tomcat/webapps/testapp

# Create the index html
echo testapp v1 > /data/tomcat/webapps/testapp/index.html

# Download Tomcat 8
wget https://mirrors.huaweicloud.com/apache/tomcat/tomcat-8/v8.5.50/bin/apache-tomcat-8.5.50.tar.gz

# Extract
tar -zxvf apache-tomcat-8.5.50.tar.gz

# Move it to /apps/tomcat
mv apache-tomcat-8.5.50/* /apps/tomcat

# Delete the archive file and the extracted empty direcotries
rm -rf apache-tomcat-8.5.50 apache-tomcat-8.5.50.tar.gz

# Edit the profile file
vim /etc/profile

# Add the following content to the end of the file:
export CATALINA_HOME=/apps/tomcat
export PATH=$PATH:$CATALINA_HOME/bin

# Apply the environment variables (Source the profile file)
source /etc/profile

# Verify
echo $CATALINA_HOME

Create the Tomcat user and set the appropriate permissions 

1
2
3

# Create the Tomcat user (Optionally, enhance the security)
useradd -s /bin/bash tomcat

# Change the owner of the Tomcat direcotry to tomcat
chown -R tomcat:tomcat /apps/tomcat

# Add execute permission to the script
chmod +x /apps/tomcat/bin/*.sh

    unpackWARs="true" autoDeploy="true">

    <!-- Map /testapp to the /data/tomcat/webapps/testapp directory -->
    <Context path="/testapp" docBase="/data/tomcat/webapps/testapp" reloadable="false" />

    ......

    <Valve className="org.apache.catalina.valves.AccessLogValve" directory="logs"
            prefix="localhost_access_log" suffix=".txt"
            pattern="%h %l %u %t &quot;%r&quot; %s %b" />
</Host>

Exit the container and package it into an image

# Exit the container without stopping it
# Pass Ctrl + P + Q
  1. method 1 : Package using docker commit (recommended for testing)
1

4
5
6
7
8

# Check the container status
docker ps

# Package the container into an image
docker commit tomcat-builder tomcat-web:app1

# Check the images
docker images
  1. method 2 : Use docker export/import (cleaner, does not preserve history).
1
2
3
4
5

# Export the container

# Import it as an image
docker import tomcat-web.tar tomcat-web:app1

# Check the logs
