"""

import json
import os

from dotenv import load_dotenv
from openai import OpenAI

load_dotenv()

# 1. 指向智谱的 OpenAI 兼容端点
client = OpenAI(
    api_key=os.getenv("ZAI_API_KEY"),
    base_url=os.getenv("ZAI_BASE_URL")
)

# 2. 通过JSON结构定义工具，包括工具名称，描述，參数等
tools = [
    {
        "type": "function",         # 类型，都是function，函数调用
        "function": {
            "name": "get_weather",          # 函数名，工具方法名相同
            "description": "获取地区特定日期的天气预报.",    # 对函数的作用进行描述
            "parameters": {
                "type": "object",       # 指定参数类型为object
                "properties": {
                    "city": {       # 第一个参数
                        "type": "string",   # 参数类型
                        "description": "城市名称，例如 北京市",
                    },
                    "date": {       # 第二个参数
                        "type": "string",   # 参数类型
                        "description": "想要查询的天气日期 YYYY-MM-dd,例如 2023-12-25"
                    }
                },
                "required": ["city", "date"],       # 调用工具的话必须回传这两个参数
                "additionalProperties": False,  # 额外的参数，不允许；限制大模型除了必传的参数，其他都不允许
            },
            "strict": True,     # 严格调用
        },
    },
]


# 工具函数，函数名与tools中相同
def get_weather(city, date):
    # 后方的天气可以调用天气预报的接口
    return f"{city} 今天 {date}天气是: 气象局播报【晴天，温度是 25 度，风向是南风。】"    # 假设来自气象局的天气数据


# 用户提示词
messages = [
    # 注意，需要联网才能返回正确的时间
    {"role":"system", "content":"如果大模型调用工具，日期必须按照格式传递： yyyy-MM-dd ；如果涉及今天，昨天，必须计算日期"},
    {"role": "user", "content": "北京 今天 日天气如何?"}
]

# 3. Prompt the model with tools defined
response = client.chat.completions.create(
    model="glm-5.3-flash",   # 或 "glm-5.3"
    messages=messages,
    tools=tools,
)

for tool_call in response.choices[0].message.tool_calls or []:
    if tool_call.function.name == "get_weather":
        # 获取大模型返回的参数
        args = json.loads(tool_call.function.arguments)
        weather = get_weather(args["city"], args["date"])

        # 4. 回传 assistant 的 tool_calls 消息（AI提示词）
        messages.append(response.choices[0].message)

        # 5. 将工具函数的执行结果添加到消息列表中（工具提示词）
        messages.append(
            {
                "role": "tool",
                "tool_call_id": tool_call.id,
                "content": json.dumps({"weather": weather}),
            }
        )

        # 6. 再次调用模型，生成最终回复
        response = client.chat.completions.create(
            model="glm-5.3-flash",
            messages=messages,
            tools=tools,
        )

        print(response.choices[0].message.content)

import os

from dotenv import load_dotenv
from langchain_core.messages import HumanMessage, ToolMessage
from langchain.tools import tool
from langchain_openai import ChatOpenAI

# 可选：通过BaseModel详细定义工具的参数
# from pydantic import BaseModel,Field
# class GetWeatherArgs(BaseModel):
#     city: str = Field(description="城市名称")
#     date: str = Field(description="日期，格式为 YYYY-MM-DD")

# 编写工具函数
@tool
def get_weather(city, date) -> str:
    """获取指定城市在指定日期的天气"""
    return f"{city} 在 {date} 天气多云，有下雨的可能"

llm = ChatOpenAI(
    model="glm-5.3-flash",
    temperature=0.0,
    api_key=os.getenv("ZAI_API_KEY"),
    base_url=os.getenv("ZAI_BASE_URL")
)

# 绑定工具函数
llm = llm.bind_tools([get_weather]) # 注意，必须重新赋值

# 定义消息列表
message_list = [
    HumanMessage(content="北京 2024-12-25 的天气")
]

res = llm.invoke(message_list)

# 当前结果当中包含调用工具的出参
print('LLM 的首次回复: ', res)

#构造 Message 对象
