List all virtual machine disk paths

cat list_all_vm_disk_path.sh
#!/bin/bash
for vm in $(virsh list --all --name | grep -v '^$'); do
    echo "=== $vm ==="
    virsh domblklist "$vm" | awk 'NR>2 && $2 != "-" {print $2}'
done
