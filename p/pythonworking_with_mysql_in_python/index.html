import pymysql

def insert_data():
    # 1.Create a connection and connect to the database
    conn = pymsql.connect(
        host = "localhost",      # host address
        port = 3306,             # port number
        user = "root",           # username
        password = "root",       # password
        database = "bj260706_3", # database name
        charset = "utf8",       
        cursorclass = pymsql.cursors.DictCursor # For query operations, convert the query results into a dictionary
    )

    # 2.Create a cursor object
    cursor = conn.cursor()
    # 3.Prepare the SQL
    sql = "insert into user(username,password) values(%s,%s)"
    # 4.Execute the SQL
    cursor.execute(sql,("tom","111"))
    # 5.Commit the transaction
    conn.commit()
    # 6.Close both the cursor and the connection
    cursor.close()
    conn.close()
    
if __name__ == '__main__':
    insert_data()

import pymysql

def query_data():
    # 1.Create a connection and connect the database
    conn = pymysql.connect(
        host="localhost",           # host address
        port=3306,                  # port number
        user="root",                # username
        password="123456",          # password
        database="bj260706_3",      # database name
        charset="utf8",
        cursorclass=pymysql.cursors.DictCursor
    )

    # 2.Create a cursor object
    cursor = conn.cursor()

    # 3.Prepare the SQL
    sql = "select * from user";

    # 4.Execute the SQL
    cursor.execute(sql)

    # 5.Fetch the query results
    result = cursor.fetchall()

    for item in result:
        print(item.get("uid"),item.get("username"),item.get("uid"))
    # 6.Close both the cursor and the connection
    cursor.close()
    conn.close()

if __name__ == '__main__':
    query_data()
